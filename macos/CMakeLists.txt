cmake_minimum_required(VERSION 3.20)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../Cargo.toml" BEANKEY_CARGO_TOML)
string(REGEX MATCH "version = \"([0-9]+\\.[0-9]+\\.[0-9]+)\"" _ "${BEANKEY_CARGO_TOML}")
project(beankey-macos VERSION "${CMAKE_MATCH_1}" LANGUAGES CXX OBJCXX)
include(CTest)
find_package(Threads REQUIRED)
if(NOT APPLE)
  message(FATAL_ERROR "The InputMethodKit frontend requires macOS")
endif()
add_subdirectory(../ipc/cpp ${CMAKE_CURRENT_BINARY_DIR}/ipc)
find_library(COCOA_FRAMEWORK Cocoa REQUIRED)
find_library(IMK_FRAMEWORK InputMethodKit REQUIRED)
find_library(CARBON_FRAMEWORK Carbon REQUIRED)
find_library(IMAGEIO_FRAMEWORK ImageIO REQUIRED)
find_library(COREGRAPHICS_FRAMEWORK CoreGraphics REQUIRED)
find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
find_library(CORETEXT_FRAMEWORK CoreText REQUIRED)
set(BEANKEY_DAEMON_PATH "" CACHE FILEPATH "Absolute daemon executable path")
set(BEANKEY_CONFIG_PATH "" CACHE FILEPATH "Internal daemon configuration path")

add_library(beankey-macos-input STATIC src/key_event.mm src/text_input.mm
  src/daemon_connection.mm src/candidate_panel.mm src/input_session.mm)
target_compile_features(beankey-macos-input PUBLIC cxx_std_20)
target_compile_options(beankey-macos-input PUBLIC -fobjc-arc PRIVATE -Wall -Wextra -Werror)
target_include_directories(beankey-macos-input PUBLIC src)
target_compile_definitions(beankey-macos-input PRIVATE
  BEANKEY_DAEMON_PATH="${BEANKEY_DAEMON_PATH}" BEANKEY_CONFIG_PATH="${BEANKEY_CONFIG_PATH}")
target_link_libraries(beankey-macos-input PUBLIC beankey-ipc ${COCOA_FRAMEWORK} ${IMK_FRAMEWORK} ${CARBON_FRAMEWORK})
add_executable(beanKey MACOSX_BUNDLE src/main.mm src/input_controller.mm)
add_executable(beankey-make-icon src/make_icon.cpp)
target_link_libraries(beankey-make-icon PRIVATE ${COREGRAPHICS_FRAMEWORK} ${IMAGEIO_FRAMEWORK} ${COREFOUNDATION_FRAMEWORK} ${CORETEXT_FRAMEWORK})
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/beanKey-character.tiff
  COMMAND beankey-make-icon ${CMAKE_CURRENT_BINARY_DIR}/beanKey-character.tiff DEPENDS beankey-make-icon)
target_sources(beanKey PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/beanKey-character.tiff)
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/beanKey-character.tiff PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
target_compile_options(beanKey PRIVATE -Wall -Wextra -Werror)
target_link_libraries(beanKey PRIVATE beankey-macos-input)
set_target_properties(beanKey PROPERTIES MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in)
install(TARGETS beanKey BUNDLE DESTINATION "Library/Input Methods")
add_executable(beankey-register-input-method src/register_input_method.mm)
target_compile_options(beankey-register-input-method PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(beankey-register-input-method PRIVATE ${COCOA_FRAMEWORK} ${CARBON_FRAMEWORK})
install(TARGETS beankey-register-input-method RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(beankey-macos-input-test tests/input_test.mm)
  target_compile_options(beankey-macos-input-test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(beankey-macos-input-test PRIVATE beankey-macos-input)
  add_test(NAME beankey-macos-input COMMAND beankey-macos-input-test)
  add_executable(beankey-macos-session-test tests/session_test.mm)
  target_compile_options(beankey-macos-session-test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(beankey-macos-session-test PRIVATE beankey-macos-input Threads::Threads)
  # AppKit integration requires a logged-in WindowServer; run outside Nix sandbox.
  option(BEANKEY_GUI_TESTS "Run AppKit integration tests in a desktop session" OFF)
  if(BEANKEY_GUI_TESTS)
    add_test(NAME beankey-macos-session COMMAND beankey-macos-session-test)
  endif()
endif()
